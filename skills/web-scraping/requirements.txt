# web-scraping skill dependencies
# Stack chosen for being lightweight, actively maintained, and
# MIT/BSD/Apache-2.0 licensed only.
#
# Install (base or research-agent conda env):
#   pip install -r requirements.txt
#
# The optional dynamic-rendering dependency (Playwright) is listed at the
# bottom and is NOT required for static / academic / file-harvest modes.

# --- core: HTTP + static HTML extraction ----------------------------------
httpx>=0.27            # HTTP client (sync + async, HTTP/2). requests successor.

beautifulsoup4>=4.12   # lenient parser for malformed HTML; final fallback.
pandas>=2.0            # pandas.read_html for HTML table extraction.

# --- academic metadata ----------------------------------------------------

# --- file harvesting / conversion ----------------------------------------
